cmake_minimum_required(VERSION 3.20)
project(ThreadScope VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

option(THREADSCOPE_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
function(threadscope_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
        if(THREADSCOPE_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(THREADSCOPE_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()

set(THREADSCOPE_SANITIZER "" CACHE STRING "Optional sanitizer: address, undefined, or thread")
if(THREADSCOPE_SANITIZER STREQUAL "address")
    add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
elseif(THREADSCOPE_SANITIZER STREQUAL "undefined")
    add_compile_options(-fsanitize=undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=undefined)
elseif(THREADSCOPE_SANITIZER STREQUAL "thread")
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
elseif(NOT THREADSCOPE_SANITIZER STREQUAL "")
    message(FATAL_ERROR "THREADSCOPE_SANITIZER must be address, undefined, thread, or empty")
endif()

add_library(threadscope_core src/scheduler.cpp src/trace.cpp src/explore.cpp)
target_include_directories(threadscope_core PUBLIC include)
target_include_directories(threadscope_core SYSTEM PRIVATE third_party)
target_link_libraries(threadscope_core PUBLIC Threads::Threads)
threadscope_warnings(threadscope_core)

add_library(threadscope_examples examples/lost_update.cpp examples/scenarios.cpp)
target_link_libraries(threadscope_examples PUBLIC threadscope_core)
target_include_directories(threadscope_examples PUBLIC examples)
threadscope_warnings(threadscope_examples)

add_executable(threadscope src/main.cpp)
target_link_libraries(threadscope PRIVATE threadscope_examples)
threadscope_warnings(threadscope)

option(THREADSCOPE_BUILD_BENCHMARKS "Build the reproducible measurement harness" ON)
if(THREADSCOPE_BUILD_BENCHMARKS)
    add_executable(threadscope_benchmark tools/benchmark.cpp)
    target_link_libraries(threadscope_benchmark PRIVATE threadscope_examples)
    target_include_directories(threadscope_benchmark SYSTEM PRIVATE third_party)
    threadscope_warnings(threadscope_benchmark)
endif()
